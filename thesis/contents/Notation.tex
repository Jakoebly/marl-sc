	\newglossaryentry{#1}{
		type=notation, % glossary type
		name={#2}, % actual notation (e.g. $\mathcal{S}$)
		description={#3}, %
		sort={#4} % string representation to determine the position in glossary overview (e.g. S)
	}
}
