% creates a new glossary
\newglossary[nlg]{notation}{not}{ntn}{Notation}

% helper command to define new notation items
\newcommand{\newnotation}[4]{
	\newglossaryentry{#1}{
		type=notation, % glossary type
		name={#2}, % actual notation (e.g. $\mathcal{S}$)
		description={#3}, %
		sort={#4} % string representation to determine the position in glossary overview (e.g. S)
	}
}

%% ===========================================================================
%% Problem-setting notation (Chapter 3).
%%
%% Symbols reserved for the reinforcement-learning formulation in the
%% Methodology chapter and therefore NOT used as physical quantities here:
%%   s, \mathcal{S}  (state, state space)
%%   a, \mathcal{A}  (action, action space)
%%   o, \mathcal{O}  (observation, observation space)
%%   r               (per-step reward signal; reward = -cost)
%%   \mathcal{R}     (reward function, if a symbol is needed; distinct from R)
%%   G               (return, cumulative reward)
%%   \pi, \theta     (policy, policy parameters)
%%   \gamma          (discount factor)
%%   V, Q            (value functions)
%%   A               (advantage)
%%   P               (transition kernel)
%%   \tau            (trajectory)
%%
%% The warehouse index i is reused intentionally as the agent index in the
%% Methodology chapter ("warehouse i is controlled by agent i").
%% ===========================================================================

%% Indices
\newnotation{idxWarehouse}{\ensuremath{i}}{index of a warehouse, reused as the index of the agent that controls it}{a-i}
\newnotation{idxRegion}{\ensuremath{j}}{index of a demand region}{a-j}
\newnotation{idxSku}{\ensuremath{k}}{index of a stock-keeping unit}{a-k}
\newnotation{idxPeriod}{\ensuremath{t}}{index of a discrete time period}{a-t}

%% Cardinalities
\newnotation{numWarehouses}{\ensuremath{W}}{number of warehouses in the network}{b-W}
\newnotation{numRegions}{\ensuremath{R}}{number of demand regions, with the assumption $R = W$ adopted throughout this thesis}{b-R}
\newnotation{numSkus}{\ensuremath{K}}{number of stock-keeping units}{b-K}
\newnotation{horizon}{\ensuremath{T}}{planning horizon, the number of periods in one episode}{b-T}

%% Network structure
\newnotation{homeRegion}{\ensuremath{h(i)}}{home region of warehouse $i$, the closest region by distance, equal to $i$ under the symmetric labelling used here}{c-h}
\newnotation{distance}{\ensuremath{d_{ij}}}{distance from warehouse $i$ to demand region $j$}{c-d}
\newnotation{skuWeight}{\ensuremath{w_{k}}}{unit weight of stock-keeping unit $k$}{c-w}
\newnotation{leadTime}{\ensuremath{L_{ik}}}{replenishment lead time for warehouse $i$ and SKU $k$, the number of periods between placing an order and its availability for fulfilment}{c-L}

%% Demand model
\newnotation{numOrders}{\ensuremath{N^{\mathrm{ord}}_{jt}}}{number of customer orders arriving in region $j$ during period $t$}{d-N}
\newnotation{rateOrders}{\ensuremath{\lambda^{\mathrm{ord}}_{j}}}{Poisson rate of customer orders per period in region $j$}{d-lambdaord}
\newnotation{probSku}{\ensuremath{p^{\mathrm{sku}}_{jk}}}{probability that SKU $k$ is included in a given order from region $j$}{d-psku}
\newnotation{rateQty}{\ensuremath{\lambda^{\mathrm{qty}}_{jk}}}{Poisson rate of the requested quantity of SKU $k$ in an order from region $j$ that includes it}{d-lambdaqty}
\newnotation{orderDemand}{\ensuremath{D^{o}_{k}}}{quantity of SKU $k$ requested by order $o$}{d-Do}
\newnotation{regionDemand}{\ensuremath{D_{jkt}}}{realised demand for SKU $k$ in region $j$ during period $t$, aggregated over all orders}{d-D}

%% State and decision variables
\newnotation{orderQty}{\ensuremath{q_{ikt}}}{replenishment order quantity placed by warehouse $i$ for SKU $k$ in period $t$}{e-q}
\newnotation{inventory}{\ensuremath{I_{ikt}}}{on-hand inventory of SKU $k$ at warehouse $i$ at the decision epoch of period $t$, always non-negative under lost sales}{e-I}
\newnotation{shipped}{\ensuremath{f_{ijkt}}}{units of SKU $k$ shipped by warehouse $i$ to region $j$ in period $t$}{e-f}
\newnotation{lostSales}{\ensuremath{U_{jkt}}}{unfulfilled (lost) demand for SKU $k$ in region $j$ during period $t$}{e-U}

%% Costs
\newnotation{costHolding}{\ensuremath{c^{\mathrm{h}}}}{holding cost per unit weight of inventory held for one period}{f-ch}
\newnotation{costPenalty}{\ensuremath{c^{\mathrm{p}}}}{penalty cost per unit of unfulfilled demand}{f-cp}
\newnotation{costOutbound}{\ensuremath{c^{\mathrm{out}}_{ij}}}{outbound shipment cost per unit weight from warehouse $i$ to region $j$}{f-cout}
\newnotation{costInbound}{\ensuremath{c^{\mathrm{in}}_{ik}}}{inbound shipment cost per unit weight of replenishment received by warehouse $i$ for SKU $k$}{f-cin}
\newnotation{periodCost}{\ensuremath{C_{t}}}{total supply-chain cost incurred in period $t$}{f-Ct}
\newnotation{totalCost}{\ensuremath{C}}{total supply-chain cost over the planning horizon, $C = \sum_{t} C_{t}$}{f-C}
